📋 Attendance Entity Database Schema

Column Name JPA Field Data Type Constraints / Notes
id Long id BIGINT PRIMARY KEY, AUTO_INCREMENT
student_id Student student BIGINT (FK) FOREIGN KEY -> students(id), NOT NULL
activity_id Activity activity BIGINT (FK) FOREIGN KEY -> activities(id), NOT NULL
faculty_id Faculty markedByFaculty BIGINT (FK) FOREIGN KEY -> faculty(id), Nullable (approver/marker)
attendance_date LocalDate attendanceDate DATE NOT NULL, INDEXED
status String status VARCHAR(30) NOT NULL, INDEXED (PRESENT, ABSENT, LATE, ON_DUTY)
session_slot String sessionSlot VARCHAR(50) NOT NULL, UNIQUE COMPONENT
remarks String remarks VARCHAR(500) Nullable (notes, leave reason, OD duty)
created_at / updated_at LocalDateTime TIMESTAMP Auditing via @PrePersist / @PreUpdate
